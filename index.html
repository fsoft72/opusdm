OpusDM apt repository

Ubuntu 26.04 and later, amd64.

sudo install -d -m 0755 /etc/apt/keyrings
sudo curl -fsSL https://fsoft72.github.io/opusdm/opusdm.gpg -o /etc/apt/keyrings/opusdm.gpg
echo "deb [signed-by=/etc/apt/keyrings/opusdm.gpg] https://fsoft72.github.io/opusdm stable main" | sudo tee /etc/apt/sources.list.d/opusdm.list
sudo apt update
sudo apt install opusdm
